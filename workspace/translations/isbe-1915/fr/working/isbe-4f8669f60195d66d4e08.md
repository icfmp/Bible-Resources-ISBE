# AZZUR
<!-- ISBE-END:1 -->

<!-- ISBE-BLOCK:2 -->
az'-ur (`azzur, « secourable ») :

<!-- ISBE-END:3 -->

<!-- ISBE-BLOCK:5 -->

<!-- ISBE-END:5 -->
